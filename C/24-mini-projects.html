🚀 Project 1: Bank Account Management System

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

#define DB_FILE "bank_records.dat"

typedef struct {
    int account_number;
    char name[32];
    double balance;
} BankAccount;

void create_account(void) {
    FILE *fp = fopen(DB_FILE, "ab"); // Append binary mode
    if (fp == NULL) { perror("Failed to open DB"); return; }

    BankAccount acc;
    printf("Enter Account Number: ");
    if (scanf("%d", &acc.account_number) != 1) { fclose(fp); return; }
    printf("Enter Customer Name  : ");
    scanf("%31s", acc.name);
    printf("Enter Initial Deposit: ");
    if (scanf("%lf", &acc.balance) != 1) { fclose(fp); return; }

    fwrite(&acc, sizeof(BankAccount), 1, fp);
    fclose(fp);
    printf("[Success] Account #%d created for %s.\n", acc.account_number, acc.name);
}

void display_all_accounts(void) {
    FILE *fp = fopen(DB_FILE, "rb");
    if (fp == NULL) { puts("No bank database records found."); return; }

    BankAccount acc;
    puts("\n=== ALL BANK ACCOUNTS ===");
    while (fread(&acc, sizeof(BankAccount), 1, fp) == 1) {
        printf("Acc #%d | Name: %-15s | Balance: $%.2lf\n",
               acc.account_number, acc.name, acc.balance);
    }
    fclose(fp);
}

int main(void) {
    int choice;
    do {
        puts("\n--- BANK MANAGEMENT SYSTEM ---");
        puts("1. Create New Account");
        puts("2. Display All Accounts");
        puts("3. Exit");
        printf("Choice: ");
        if (scanf("%d", &choice) != 1) break;

        if (choice == 1) create_account();
        else if (choice == 2) display_all_accounts();
    } while (choice != 3);

    puts("Exiting Bank System. Goodbye!");
    return 0;
}

🚀 Project 2: Terminal Command Line Parser

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_LINE 256
#define MAX_TOKENS 16

int parse_command_line(char *input_line, char *tokens[]) {
    int token_count = 0;
    char *tok = strtok(input_line, " \t\r\n");

    while (tok != NULL && token_count < MAX_TOKENS - 1) {
        tokens[token_count++] = tok;
        tok = strtok(NULL, " \t\r\n");
    }
    tokens[token_count] = NULL; // Null-terminate argument list for exec
    return token_count;
}

int main(void) {
    char line[MAX_LINE];
    char *tokens[MAX_TOKENS];

    puts("=== SIMPLE C COMMAND PARSER (Type 'exit' to quit) ===");
    while (1) {
        printf("c_shell> ");
        if (fgets(line, sizeof(line), stdin) == NULL) break;

        int count = parse_command_line(line, tokens);
        if (count == 0) continue;

        if (strcmp(tokens[0], "exit") == 0) break;

        printf("Parsed Command: '%s' (%d arguments)\n", tokens[0], count - 1);
        for (int i = 0; i < count; i++) {
            printf("  arg[%d] = \"%s\"\n", i, tokens[i]);
        }
    }
    return 0;
}

⚠️ Common Mistakes & Pitfalls

  • Mixing Interactive scanf() and fgets(): Calling scanf("%d") leaves trailing \n in the input buffer, causing subsequent fgets() calls to read an empty line immediately! Clean buffer using getchar().
  • Hardcoding Fixed Array Bounds: Assuming terminal inputs or database record counts will fit in tiny fixed buffers.
