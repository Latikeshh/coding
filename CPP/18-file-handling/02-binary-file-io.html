Binary File I/O (read & write)

🟡 Intermediate

📖 Definition

Opening files with std::ios::binary flag enables direct binary transfer of POD struct memory to disk using file.write(reinterpret_cast<char*>(&obj), sizeof(obj)) and file.read(...).

💡 Code Example

#include <iostream>
#include <fstream>

struct Record {
    int id;
    double score;
};

int main() {
    Record r1{101, 98.5};

    // Write binary
    std::ofstream out("data.bin", std::ios::binary);
    out.write(reinterpret_cast<char*>(&r1), sizeof(r1));
    out.close();

    // Read binary
    Record r2{};
    std::ifstream in("data.bin", std::ios::binary);
    in.read(reinterpret_cast<char*>(&r2), sizeof(r2));

    std::cout << "Read Record ID: " << r2.id << ", Score: " << r2.score << "\n";
    return 0;
}
