💡 Code Example

#include <iostream>

int main() {
    int age = 25;
    double price = 19.99;
    char grade = 'A';
    bool passed = true;

    std::cout << "Age: " << age << ", Price: $" << price << ", Grade: " << grade << ", Passed: " << std::boolalpha << passed << "\n";
    return 0;
}

👀 Output

Age: 25, Price: $19.99, Grade: A, Passed: true
