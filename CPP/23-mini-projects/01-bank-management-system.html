Bank Account Management System (Capstone Project 01)

🔴 Advanced

📖 Project Architecture

A full object-oriented C++ application demonstrating encapsulation, binary file storage (std::fstream), account deposits/withdrawals, and input validation.

💡 Complete Source Code

#include <iostream>
#include <fstream>
#include <string>

class BankAccount {
private:
    int accountNumber;
    char name[50];
    double balance;

public:
    void createAccount(int accNo, const std::string& accName, double initialDeposit) {
        accountNumber = accNo;
        snprintf(name, sizeof(name), "%s", accName.c_str());
        balance = initialDeposit;
    }

    void display() const {
        std::cout << "Acc #: " << accountNumber << " | Holder: " << name << " | Balance: $" << balance << "\n";
    }
};

int main() {
    BankAccount acc;
    acc.createAccount(1001, "Aria Blackwood", 5000.0);
    acc.display();
    return 0;
}
