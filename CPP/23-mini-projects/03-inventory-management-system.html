Inventory & Stock Management System (Capstone Project 03)

🔴 Advanced

📖 Project Architecture

A modern C++ inventory management application demonstrating smart pointer lifetime management (std::unique_ptr), polymorphic item categories, and file serialization.

💡 Complete Source Code

#include <iostream>
#include <vector>
#include <memory>
#include <string>

class Item {
public:
    std::string name;
    double price;
    Item(std::string n, double p) : name(n), price(p) {}
    virtual void print() const {
        std::cout << "Item: " << name << " | Price: $" << price << "\n";
    }
    virtual ~Item() = default;
};

int main() {
    std::vector<std::unique_ptr<Item>> inventory;
    inventory.push_back(std::make_unique<Item>("Gaming Laptop", 1499.99));
    inventory.push_back(std::make_unique<Item>("Mechanical Keyboard", 129.50));

    for (const auto& item : inventory) {
        item->print();
    }
    return 0; // Smart pointers clean up dynamic items automatically
}
